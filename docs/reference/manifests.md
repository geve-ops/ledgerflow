```yaml
# S3-compatible object store for database backups (stands in for S3/GCS/Azure Blob locally).
# Single replica with a persistent volume: fine for a local demo, not a durable backup target;
# in production, point the ObjectStore at a bucket in a different failure domain.
apiVersion: v1
kind: ServiceAccount
metadata:
  name: seaweedfs
  namespace: backup-storage
automountServiceAccountToken: false
---
apiVersion: v1
kind: Service
metadata:
  name: seaweedfs
  namespace: backup-storage
spec:
  selector: {app.kubernetes.io/name: seaweedfs}
  ports:
    - {name: s3, port: 8333, targetPort: s3}
---
apiVersion: apps/v1
kind: StatefulSet
metadata:
  name: seaweedfs
  namespace: backup-storage
spec:
  serviceName: seaweedfs
  replicas: 1
  selector:
    matchLabels: {app.kubernetes.io/name: seaweedfs}
  template:
    metadata:
      labels: {app.kubernetes.io/name: seaweedfs}
    spec:
      serviceAccountName: seaweedfs
      automountServiceAccountToken: false
      securityContext:
        runAsNonRoot: true
        runAsUser: 10001
        runAsGroup: 10001
        fsGroup: 10001
        seccompProfile: {type: RuntimeDefault}
      containers:
        - name: seaweedfs
          image: chrislusf/seaweedfs:4.47
          command: ["weed"]
          args:
            - server
            - -dir=/data
            - -s3
            - -s3.port=8333
            - -s3.config=/etc/seaweedfs/s3.json
            - -master.volumeSizeLimitMB=256
            - -volume.max=20
          ports:
            - {name: s3, containerPort: 8333}
          securityContext:
            allowPrivilegeEscalation: false
            capabilities: {drop: ["ALL"]}
          resources:
            requests: {cpu: 50m, memory: 256Mi}
            limits: {memory: 768Mi}
          readinessProbe:
            tcpSocket: {port: s3}
            periodSeconds: 5
          livenessProbe:
            tcpSocket: {port: s3}
            periodSeconds: 15
            failureThreshold: 6
          volumeMounts:
            - {name: data, mountPath: /data}
            - {name: s3-config, mountPath: /etc/seaweedfs, readOnly: true}
      volumes:
        - name: s3-config
          secret: {secretName: seaweedfs-s3}
  volumeClaimTemplates:
    - metadata:
        name: data
      spec:
        accessModes: ["ReadWriteOnce"]
        resources:
          requests: {storage: 5Gi}
---
# Creates the bucket once the store is up (idempotent). Runs after every sync.
apiVersion: batch/v1
kind: Job
metadata:
  name: create-backup-bucket
  namespace: backup-storage
  annotations:
    argocd.argoproj.io/hook: PostSync
    argocd.argoproj.io/hook-delete-policy: BeforeHookCreation
spec:
  backoffLimit: 12
  template:
    spec:
      restartPolicy: OnFailure
      automountServiceAccountToken: false
      securityContext:
        runAsNonRoot: true
        runAsUser: 10001
        seccompProfile: {type: RuntimeDefault}
      containers:
        - name: aws
          image: amazon/aws-cli:2.37.12
          command: ["sh", "-c"]
          args:
            - |
              export AWS_EC2_METADATA_DISABLED=true HOME=/tmp
              until aws --endpoint-url http://seaweedfs:8333 s3api head-bucket --bucket cnpg-backups 2>/dev/null \
                 || aws --endpoint-url http://seaweedfs:8333 s3 mb s3://cnpg-backups; do
                echo "waiting for object store..."; sleep 5
              done
              echo "bucket ready"
          env:
            - name: AWS_ACCESS_KEY_ID
              valueFrom: {secretKeyRef: {name: s3-credentials, key: ACCESS_KEY_ID}}
            - name: AWS_SECRET_ACCESS_KEY
              valueFrom: {secretKeyRef: {name: s3-credentials, key: ACCESS_SECRET_KEY}}
            - name: AWS_DEFAULT_REGION
              value: us-east-1
          securityContext:
            allowPrivilegeEscalation: false
            capabilities: {drop: ["ALL"]}
          resources:
            requests: {cpu: 20m, memory: 64Mi}
            limits: {memory: 256Mi}
```

# Where CloudNativePG ships base backups and WAL (continuous archiving -> point-in-time recovery).
apiVersion: barmancloud.cnpg.io/v1
kind: ObjectStore
metadata:
  name: ledger-store
  namespace: ledger
spec:
  retentionPolicy: "7d"
  configuration:
    destinationPath: s3://cnpg-backups/
    endpointURL: http://seaweedfs.backup-storage.svc.cluster.local:8333
    s3Credentials:
      accessKeyId: {name: s3-credentials, key: ACCESS_KEY_ID}
      secretAccessKey: {name: s3-credentials, key: ACCESS_SECRET_KEY}
    wal: {compression: gzip}
    data: {compression: gzip}
  instanceSidecarConfiguration:
    env:
      # Newer AWS SDK defaults add checksum headers that non-AWS S3 servers may reject.

      - {name: AWS_RESPONSE_CHECKSUM_VALIDATION, value: when_required}

  podSelector: {}
  policyTypes: [Ingress, Egress]
